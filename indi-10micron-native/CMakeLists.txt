# Keep unit-test logging/configuration/park files out of the operator's home.
# Integration helpers additionally create isolated homes for each child process.
if(BUILD_TESTING)
 get_property(driver_tests DIRECTORY PROPERTY TESTS)
 foreach(driver_test IN LISTS driver_tests)
  set(test_home "${CMAKE_CURRENT_BINARY_DIR}/test-home/${driver_test}")
  file(MAKE_DIRECTORY "${test_home}/.indi")
  set_tests_properties(${driver_test} PROPERTIES ENVIRONMENT "HOME=${test_home}")
 endforeach()
endif()
