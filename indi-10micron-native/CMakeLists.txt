cmake_minimum_required(VERSION 3.18)
project(indi_10micron_native VERSION 0.19 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(tenmicron_protocol tenmicron_protocol.cpp)
target_include_directories(tenmicron_protocol PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(tenmicron_protocol PUBLIC Threads::Threads)
target_compile_options(tenmicron_protocol PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tenmicron-probe probe.cpp)
target_link_libraries(tenmicron-probe PRIVATE tenmicron_protocol)
option(BUILD_TESTING "Build loopback and unit tests" OFF)
include(CTest)
if(BUILD_TESTING)
 add_executable(parser-test tests/parser.cpp)
 target_link_libraries(parser-test PRIVATE tenmicron_protocol)
 target_compile_options(parser-test PRIVATE -UNDEBUG)
 add_test(NAME parser COMMAND parser-test)
endif()

option(BUILD_INDI_DRIVER "Build INDI telescope drivers" ON)
if(BUILD_INDI_DRIVER)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(INDI REQUIRED IMPORTED_TARGET libindi>=2.2.1)
 find_library(INDI_DRIVER_LIBRARY NAMES indidriver REQUIRED)
 add_executable(indi_10micron_native_ro tenmicron_driver.cpp)
 target_link_libraries(indi_10micron_native_ro PRIVATE tenmicron_protocol PkgConfig::INDI ${INDI_DRIVER_LIBRARY})
 target_compile_options(indi_10micron_native_ro PRIVATE -Wall -Wextra -Wpedantic)
 install(TARGETS indi_10micron_native_ro RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
 install(FILES indi_10micron_native_ro.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/indi)
endif()
